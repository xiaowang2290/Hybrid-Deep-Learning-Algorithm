import os
import numpy as np
import matplotlib.pyplot as plt
import torch
import torch.nn as nn
import torch.optim as optim
import warnings

warnings.filterwarnings("ignore")
plt.rcParams["font.sans-serif"] = ["DejaVu Sans"]
plt.rcParams["axes.unicode_minus"] = False

torch.set_default_dtype(torch.float64)

# ===================== 1) 4D Hyperchaotic ODE =====================
def ode_rhs(t, s, a=36.0, b=3.0, c=20.0, d=1.0):
    x, y, z, w = s
    dx = a * (y - x) + w
    dy = (c - a) * x - x * z + c * y
    dz = x * y - b * z
    dw = x * z - d * w
    return np.array([dx, dy, dz, dw], dtype=np.float64)


# ===================== 2) PHeun / Improved Euler =====================
def imp_euler_step_vec(ti, si, h, a, b, c, d):
    k1 = ode_rhs(ti, si, a, b, c, d)
    k2 = ode_rhs(ti + h, si + h * k1, a, b, c, d)
    s_euler = si + (h / 2.0) * (k1 + k2)
    return s_euler, (k1, k2)


# ===================== 3) Correction Net =====================
class CorrectionNet(nn.Module):
    def __init__(self, input_dim=15, hidden_dim=256, depth=4, out_dim=4):
        super().__init__()
        layers = [nn.Linear(input_dim, hidden_dim), nn.GELU()]
        for _ in range(depth - 1):
            layers += [nn.Linear(hidden_dim, hidden_dim), nn.GELU()]
        layers += [nn.Linear(hidden_dim, out_dim)]
        self.net = nn.Sequential(*layers)

    def forward(self, x):
        return self.net(x)


# ===================== 4) Trainer: only for PHeun, h=0.001 =====================
class PHeunTrainer:
    def __init__(self, a=36.0, b=3.0, c=20.0, d=1.0, h=1e-3, t_max=8.0, device=None, seed=0):
        np.random.seed(seed)
        torch.manual_seed(seed)

        self.a, self.b, self.c, self.d = float(a), float(b), float(c), float(d)
        self.h, self.t_max = float(h), float(t_max)
        self.device = torch.device(device if device else ("cuda" if torch.cuda.is_available() else "cpu"))

        self.model = CorrectionNet(input_dim=15, hidden_dim=256, depth=4, out_dim=4).to(self.device).double()
        self.mean = None
        self.std = None

    def _make_one_step_dataset(self, s0_list):
        h = self.h
        t_grid = np.arange(0.0, self.t_max + 1e-15, h)
        inputs, targets = [], []

        for s0 in s0_list:
            S_traj = np.zeros((len(t_grid), 4), dtype=np.float64)
            S_traj[0] = s0

            for i in range(len(t_grid) - 1):
                ti = t_grid[i]
                si = S_traj[i]

                # PHeun一步
                s_h, (k1, k2) = imp_euler_step_vec(ti, si, h, self.a, self.b, self.c, self.d)
                S_traj[i + 1] = s_h

                # 两个半步
                s_half_1, _ = imp_euler_step_vec(ti, si, h / 2, self.a, self.b, self.c, self.d)
                s_h2, _ = imp_euler_step_vec(ti + h / 2, s_half_1, h / 2, self.a, self.b, self.c, self.d)

                # Richardson误差估计
                e_est = (s_h2 - s_h) / 3.0

                feat = np.concatenate([
                    np.array([ti, ti + h], dtype=np.float64),
                    si.astype(np.float64),
                    k1.astype(np.float64),
                    k2.astype(np.float64),
                    np.array([h], dtype=np.float64)
                ], axis=0)

                target_val = e_est / (h ** 3)

                inputs.append(feat)
                targets.append(target_val.astype(np.float64))

        X = np.stack(inputs, axis=0)
        Y = np.stack(targets, axis=0)

        mean = X.mean(axis=0, keepdims=True)
        std = X.std(axis=0, keepdims=True)
        std[std == 0] = 1.0

        self.mean = torch.tensor(mean, dtype=torch.float64, device=self.device)
        self.std = torch.tensor(std, dtype=torch.float64, device=self.device)

        X_t = torch.tensor(X, dtype=torch.float64, device=self.device)
        Y_t = torch.tensor(Y, dtype=torch.float64, device=self.device)
        Xn = (X_t - self.mean) / self.std

        return Xn, Y_t

    def train(self, epochs=3000, lr=5e-4):
        base = np.array([1.0, 1.0, 1.0, 1.0], dtype=np.float64)
        s0_list = [
            base,
            base * 0.5,
            base * 1.5,
            base + np.array([0.2, -0.1, 0.1, -0.2]),
            base + np.array([-0.3, 0.2, -0.2, 0.3]),
        ]

        Xn, Y = self._make_one_step_dataset(s0_list)

        opt = optim.AdamW(self.model.parameters(), lr=lr, weight_decay=1e-6)
        sched = optim.lr_scheduler.CosineAnnealingLR(opt, T_max=epochs)
        loss_fn = nn.MSELoss()

        self.model.train()
        for ep in range(1, epochs + 1):
            opt.zero_grad(set_to_none=True)
            pred = self.model(Xn)
            loss = loss_fn(pred, Y)
            loss.backward()
            opt.step()
            sched.step()

            if ep == 1 or ep % 500 == 0:
                print(f"[PHEUN h={self.h}] Epoch {ep:4d} | Loss: {loss.item():.3e}")

        return self

    def save_model(self, path):
        checkpoint = {
            "model_state": self.model.state_dict(),
            "mean": self.mean,
            "std": self.std
        }
        torch.save(checkpoint, path)

    def load_model(self, path):
        checkpoint = torch.load(path, map_location=self.device)
        self.model.load_state_dict(checkpoint["model_state"])
        self.mean = checkpoint["mean"].to(self.device)
        self.std = checkpoint["std"].to(self.device)
        self.model.eval()

    @torch.no_grad()
    def predict_N(self, feat_arr):
        self.model.eval()
        feat = torch.tensor(feat_arr, dtype=torch.float64, device=self.device).view(1, -1)
        feat_n = (feat - self.mean) / self.std
        out = self.model(feat_n).cpu().numpy().reshape(-1)
        return out.astype(np.float64)


# ===================== 5) Corrected Simulation =====================
def simulate_corrected_pheun(trainer, s0, t_final):
    a, b, c, d, h = trainer.a, trainer.b, trainer.c, trainer.d, trainer.h
    t_grid = np.arange(0.0, t_final + 1e-15, h)
    S = np.zeros((len(t_grid), 4), dtype=np.float64)
    S[0] = np.array(s0, dtype=np.float64)

    for i in range(len(t_grid) - 1):
        ti = t_grid[i]
        si = S[i]

        s_base, (k1, k2) = imp_euler_step_vec(ti, si, h, a, b, c, d)

        feat = np.concatenate([
            np.array([ti, ti + h], dtype=np.float64),
            si.astype(np.float64),
            k1.astype(np.float64),
            k2.astype(np.float64),
            np.array([h], dtype=np.float64)
        ], axis=0)

        N_out = trainer.predict_N(feat)
        S[i + 1] = s_base + N_out * (h ** 3)

    return t_grid, S


# ===================== 6) Get or Train =====================
def get_or_train_pheun_model(h=0.001, t_train=8.0):
    save_path = f"model_pheun_h{h}.pth"
    trainer = PHeunTrainer(h=h, t_max=t_train)

    if os.path.exists(save_path):
        print(f"Loading existing model: {save_path}")
        trainer.load_model(save_path)
    else:
        print(f"Training new model: {save_path} ...")
        trainer.train(epochs=3000, lr=5e-4)
        trainer.save_model(save_path)

    return trainer
def plot_phase_portraits_2x3(S, save_path="pheun_h0.001_phase_2x3.png"):
    names = ["x", "y", "z", "w"]
    pairs = [
        (0, 1), (0, 2), (0, 3),
        (1, 2), (1, 3), (2, 3)
    ]

    plt.figure(figsize=(14, 8))

    for idx, (i, j) in enumerate(pairs, start=1):
        plt.subplot(2, 3, idx)
        plt.plot(S[:, i], S[:, j], lw=0.8)
        plt.xlabel(names[i])
        plt.ylabel(names[j])
        plt.title(f"{names[i]}-{names[j]}")
        plt.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig(save_path, dpi=300)
    plt.show()
    print(f"Done! 2D phase portraits saved to '{save_path}'")


# ===================== 7) Main =====================
def main():
    h = 0.001
    t_train = 8.0
    t_final = 150.0
    s0 = [1.0, 1.0, 1.0, 1.0]

    trainer = get_or_train_pheun_model(h=h, t_train=t_train)

    print("\nSimulating corrected PHeun trajectory...")
    t, S = simulate_corrected_pheun(trainer, s0, t_final)

    plot_phase_portraits_2x3(S, save_path="pheun_h0.001_phase_2x3.png")

    names = ["x", "y", "z", "w"]
    plt.figure(figsize=(12, 8))
    for i in range(4):
        plt.subplot(2, 2, i + 1)
        plt.plot(t, S[:, i], lw=1.2)
        plt.title(f"{names[i]}(t)")
        plt.xlabel("t")
        plt.ylabel(names[i])
        plt.grid(True, alpha=0.3)

    plt.tight_layout()
    plt.savefig("pheun_h0.001_trajectory.png", dpi=300)

    print("Done! Plot saved to 'pheun_h0.001_trajectory.png'")


if __name__ == "__main__":
    main()
